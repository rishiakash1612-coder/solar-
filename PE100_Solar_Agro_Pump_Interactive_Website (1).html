Arduino Firmware (C++)

/* PE100 – Solar Agricultural Water Pumping System */
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

const int SOIL_PIN = A0;
const int TRIG_PIN = 9;
const int ECHO_PIN = 10;
const int RELAY_PIN = 7;

void setup() {
  Serial.begin(9600);
  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, HIGH); // OFF
}

void loop() {
  int rawSoil = analogRead(SOIL_PIN);
  int soilPercent = constrain(map(rawSoil, 800, 400, 0, 100), 0, 100);

  if (soilPercent < 35) {
    digitalWrite(RELAY_PIN, LOW); // Turn ON pump
  } else if (soilPercent >= 65) {
    digitalWrite(RELAY_PIN, HIGH); // Turn OFF pump
  }
  delay(600);
}

Top Viva Voce Questions

Q1: Why use 12V DC instead of 230V AC?
A: 12V DC is touch-safe (zero shock risk in lab) and eliminates the inverter that wastes 25% energy.


Q2: What is Dry-Run Protection?
A: Shuts pump OFF when water drops below 15% to prevent motor burnout.


Q3: Why choose a Capacitive Soil Sensor?
A: Has no exposed copper prongs so it doesn't corrode from electrolysis like resistive sensors.
